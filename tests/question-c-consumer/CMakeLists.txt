cmake_minimum_required(VERSION 3.31)
project(QuestionCConsumer LANGUAGES C CXX)
find_package(iiMultimodalAgent CONFIG REQUIRED)
if(NOT EXISTS "${IIMULTIMODALAGENT_PYTHON_DIR}/iiquestion_presentation.py")
    message(FATAL_ERROR "The staged agent package did not advertise the question Python API")
endif()
enable_testing()
add_executable(QuestionCConsumer main.c)
set_target_properties(QuestionCConsumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES LINKER_LANGUAGE CXX)
target_link_libraries(QuestionCConsumer PRIVATE iiMultimodalAgent::iiMultimodalAgent)
add_test(NAME Question.CConsumer COMMAND QuestionCConsumer)
set_tests_properties(Question.CConsumer PROPERTIES TIMEOUT 30)
if(APPLE)
    set_tests_properties(Question.CConsumer PROPERTIES ENVIRONMENT_MODIFICATION
        "DYLD_LIBRARY_PATH=unset:;DYLD_FALLBACK_LIBRARY_PATH=unset:;DYLD_FRAMEWORK_PATH=unset:;DYLD_FALLBACK_FRAMEWORK_PATH=unset:")
endif()
